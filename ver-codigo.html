package excepciones;

/**
 * Excepción de dominio lanzada cuando un dato de entrada no cumple las reglas
 * de validación (rango de año, texto vacío, precio negativo, etc.).
 *
 * <p><b>Decisión de diseño:</b> es una excepción <i>checked</i> para obligar a
 * la capa de vista a informar el error al usuario y a la capa de servicio a
 * decidir si la operación se aborta.</p>
 */
public class DatoInvalidoException extends Exception {

    /** Versión serializable de la excepción. */
    private static final long serialVersionUID = 1L;

    /**
     * Crea la excepción con un mensaje descriptivo.
     *
     * @param mensaje detalle del dato inválido
     */
    public DatoInvalidoException(String mensaje) {
        super(mensaje);
    }

    /**
     * Crea la excepción con un mensaje y la causa original.
     *
     * @param mensaje detalle del dato inválido
     * @param causa   excepción de bajo nivel que originó el error
     */
    public DatoInvalidoException(String mensaje, Throwable causa) {
        super(mensaje, causa);
    }
}

import excepciones.DatoInvalidoException;

/**
 * Automóvil de pasajeros. Especifica carrocería y cantidad de puertas, y

public class Automovil extends Vehiculo implements Serializable {

    /** Versión serializable de la clase. */
    private static final long serialVersionUID = 1L;

    /** Impuesto fijo base. */
    private static final Double IMPUESTO_BASE = 60.0;

    /** Alícuota aplicada al precio. */
    private static final Double ALICUOTA = 0.015;

    /** Recargo para vehículos fabricados a partir de 2020. */
    private static final Double RECARGO_MODERNO = 40.0;

    /** Año a partir del cual se aplica el recargo. */
    private static final Integer ANIO_RECARGO = 2020;

    /** Tipo de carrocería. */
    private TipoCarroceria carroceria;

    /** Cantidad de puertas. */
    private Integer puertas;

    /**
     * Constructor vacío requerido por el estándar JavaBean.
     */
    public Automovil() {
        super();
    }

    /**
     * Constructor parametrizado completo.
     *
     * @param identificador identificador único
     * @param marca         marca
     * @param modelo        modelo
     * @param anio          año de fabricación
     * @param color         color
     * @param estado        estado comercial
     * @param kilometraje   kilometraje
     * @param precio        precio de venta
     * @param carroceria    tipo de carrocería
     * @param puertas       cantidad de puertas
     */
    public Automovil(String identificador, String marca, String modelo, Integer anio,
                     String color, EstadoVehiculo estado, Double kilometraje, Double precio,
                     TipoCarroceria carroceria, Integer puertas) {
        super(identificador, marca, modelo, anio, color, estado, kilometraje, precio);
        this.carroceria = carroceria;
        this.puertas = puertas;
    }

    /**
     * Devuelve la carrocería.
     *
     * @return tipo de carrocería
     */
    public TipoCarroceria getCarroceria() {
        return carroceria;
    }

    /**
     * Establece la carrocería.
     *
     * @param carroceria tipo de carrocería
     */
    public void setCarroceria(TipoCarroceria carroceria) {
        this.carroceria = carroceria;
    }

    /**
     * Devuelve la cantidad de puertas.
     *
     * @return cantidad de puertas
     */
    public Integer getPuertas() {
        return puertas;
    }

    /**
     * Establece la cantidad de puertas.
     *
     * @param puertas cantidad de puertas
     */
    public void setPuertas(Integer puertas) {
        this.puertas = puertas;
    }

    /**
     * {@inheritDoc}
     */
    @Override
    public TipoVehiculo obtenerTipo() {
        return TipoVehiculo.AUTOMOVIL;
    }

    /**
     * {@inheritDoc}
     */
    @Override
    protected void validarReglasEspecificas() throws DatoInvalidoException {
        if (carroceria == null) {
            throw new DatoInvalidoException("La carrocería del automóvil es obligatoria.");
        }
        if (puertas == null || puertas < 2 || puertas > 5) {
            throw new DatoInvalidoException("La cantidad de puertas debe estar entre 2 y 5.");
        }
    }

    /**
     * Calcula el impuesto: 60 + 1,5% del precio + 40 si es del año 2020 o posterior.
     *
     * @return monto del impuesto anual de patente
     */
    @Override
    public Double calcularImpuestoPatente() {
        Double recargo = getAnio() >= ANIO_RECARGO ? RECARGO_MODERNO : 0.0;
        return redondear(IMPUESTO_BASE + getPrecio() * ALICUOTA + recargo);
    }

    /**
     * Devuelve la ficha técnica extendida del automóvil.
     *
     * @return ficha técnica legible
     */
    @Override
    public String obtenerFichaTecnicaDetallada() {
        return "AUTOMÓVIL | " + getMarca() + " " + getModelo() + " " + getAnio()
                + " | Carrocería: " + carroceria.getEtiqueta()
                + " | Puertas: " + puertas
                + " | " + getColor()
                + " | " + getEstado().getEtiqueta()
                + " | " + String.format(Locale.US, "%.0f", getKilometraje()) + " km"
                + " | Precio: $" + String.format(Locale.US, "%.2f", getPrecio());
    }

    /**
     * {@inheritDoc}
     */
    @Override
    public String toString() {
        return "Automovil{" + super.toString()
                + ", carroceria=" + carroceria
                + ", puertas=" + puertas + '}';
    }
}

/**
 * Clase abstracta base de todos los vehículos de la concesionaria.
 *
 * <p>Define la identidad de negocio (identificador único), los atributos
 * comunes, las reglas de validación compartidas y el contrato polimórfico que

    /**
     * Valida los atributos comunes, incluida la coherencia estado/kilometraje.
     *
     * @throws DatoInvalidoException si falta un dato obligatorio o está fuera de rango
     */
    private void validarDatosComunes() throws DatoInvalidoException {
        if (identificador == null || identificador.isBlank()) {
            throw new DatoInvalidoException("El identificador es obligatorio.");
        }
        if (marca == null || marca.isBlank()) {
            throw new DatoInvalidoException("La marca es obligatoria.");
        }
        if (modelo == null || modelo.isBlank()) {
            throw new DatoInvalidoException("El modelo es obligatorio.");
        }
        if (color == null || color.isBlank()) {
            throw new DatoInvalidoException("El color es obligatorio.");
        }
        if (anio == null || anio < ANIO_MINIMO || anio > ANIO_MAXIMO) {
            throw new DatoInvalidoException("El año debe estar entre " + ANIO_MINIMO + " y " + ANIO_MAXIMO + ".");
        }
        if (precio == null || precio <= 0) {
            throw new DatoInvalidoException("El precio debe ser mayor que cero.");
        }
        if (estado == null) {
            throw new DatoInvalidoException("El estado es obligatorio.");
        }
        if (kilometraje == null || kilometraje < 0) {
            throw new DatoInvalidoException("El kilometraje no puede ser negativo ni nulo.");
        }
        if (estado == EstadoVehiculo.NUEVO && kilometraje != 0) {
            throw new KilometrajeInconsistenteException(
                    "Un vehículo NUEVO debe tener 0 km; se informaron " + kilometraje + " km.");
        }
        if (estado == EstadoVehiculo.USADO && kilometraje <= 0) {
            throw new KilometrajeInconsistenteException(
                    "Un vehículo USADO debe tener más de 0 km; se informaron " + kilometraje + " km.");
        }
    }

    /**
     * Valida los atributos propios de cada tipo de vehículo.
     *
     * @throws DatoInvalidoException si algún atributo específico es inválido
     */
    protected abstract void validarReglasEspecificas() throws DatoInvalidoException;

    /**
     * Redondea un importe a dos decimales.
     *
     * @param valor importe a redondear
     * @return importe redondeado
     */
    protected static Double redondear(Double valor) {
        return Math.round(valor * 100.0) / 100.0;
    }

    /**
     * {@inheritDoc}
     *
     * <p>El precio final de venta es el precio comercial. Las subclases pueden
     * enriquecerlo con ajustes propios mediante polimorfismo.</p>
     */
    @Override
    public Double calcularPrecioFinalVenta() {
        return redondear(precio);
    }

    /**
     * {@inheritDoc}
     */
    @Override
    public Double aplicarDescuento(Double porcentaje) {
        if (porcentaje == null || porcentaje < 0 || porcentaje > 100) {
            throw new IllegalArgumentException("El porcentaje de descuento debe estar entre 0 y 100.");
        }
        return redondear(calcularPrecioFinalVenta() * (1 - porcentaje / 100.0));
    }

    /**
     * {@inheritDoc}
     */
    @Override
    public Double calcularValorTasado() {
        return redondear(precio * FACTOR_TASACION);
    }

    /**
     * Compara vehículos por su identificador único de negocio.
     *
     * @param objeto objeto a comparar
     * @return {@code true} si ambos son vehículos con el mismo identificador
     */
    @Override
    public boolean equals(Object objeto) {
        if (this == objeto) {
            return true;
        }
        if (!(objeto instanceof Vehiculo)) {
            return false;
        }
        Vehiculo otro = (Vehiculo) objeto;
        return Objects.equals(identificador, otro.identificador);
    }

    /**
     * Genera el código de hash a partir del identificador único.
     *
     * @return código de hash
     */
    @Override
    public int hashCode() {
        return Objects.hash(identificador);
    }

    /**
     * Devuelve una representación ordenada de los atributos comunes.
     *
     * @return representación textual del vehículo
     */
    @Override
    public String toString() {
        return "Vehiculo{identificador='" + identificador + '\''
                + ", marca='" + marca + '\''
                + ", modelo='" + modelo + '\''
                + ", anio=" + anio
                + ", color='" + color + '\''
                + ", estado=" + estado
                + ", kilometraje=" + kilometraje
                + ", precio=" + precio + '}';
    }
}

import excepciones.DatoInvalidoException;

import util.Validador;

import java.nio.charset.StandardCharsets;
import java.util.Base64;
import java.util.StringJoiner;

/**
 * Codificador y decodificador de vehículos en formato CSV delimitado por '|'.
 *
 * <p><b>Decisión de diseño:</b> los campos de texto se codifican en Base64 URL
 * para que el delimitador, los saltos de línea y los acentos nunca corrompan
 * un registro. El primer campo indica el tipo concreto, lo que permite
 * reconstruir polimórficamente la subclase correcta al leer el archivo.</p>
 */
public final class CsvVehiculoCodec {

    /** Delimitador de campos. */
    private static final String SEPARADOR = "\\|";

    /** Delimitador usado al escribir. */
    private static final String SEPARADOR_SALIDA = "|";

    /** Cantidad de campos comunes antes de los atributos específicos. */
    private static final int CAMPOS_COMUNES = 9;

    /** Constructor privado de clase utilitaria. */
    private CsvVehiculoCodec() {
    }

    /**
     * Serializa un vehículo en una línea CSV.
     *
     * @param vehiculo vehículo a serializar
     * @return línea CSV con el tipo y todos sus atributos
     */
    public static String serialize(Vehiculo vehiculo) {
        StringJoiner fila = new StringJoiner(SEPARADOR_SALIDA);
        fila.add(vehiculo.obtenerTipo().name());
        fila.add(codificar(vehiculo.getIdentificador()));
        fila.add(codificar(vehiculo.getMarca()));
        fila.add(codificar(vehiculo.getModelo()));
        fila.add(String.valueOf(vehiculo.getAnio()));
        fila.add(codificar(vehiculo.getColor()));
        fila.add(vehiculo.getEstado().name());
        fila.add(String.valueOf(vehiculo.getKilometraje()));
        fila.add(String.valueOf(vehiculo.getPrecio()));

        if (vehiculo instanceof Automovil) {
            Automovil auto = (Automovil) vehiculo;
            fila.add(auto.getCarroceria().name());
            fila.add(String.valueOf(auto.getPuertas()));
        } else if (vehiculo instanceof Camioneta) {
            Camioneta camioneta = (Camioneta) vehiculo;
            fila.add(camioneta.getCabina().name());
            fila.add(camioneta.getTraccion().name());
            fila.add(String.valueOf(camioneta.getCapacidadCarga()));
        } else if (vehiculo instanceof Motocicleta) {
            Motocicleta moto = (Motocicleta) vehiculo;
            fila.add(moto.getEstilo().name());
            fila.add(String.valueOf(moto.getCilindrada()));
        }
        return fila.toString();
    }

    /**
     * Reconstruye un vehículo a partir de una línea CSV y valida su contenido.
     *
     * @param linea línea CSV leída del archivo
     * @return vehículo reconstruido
     * @throws DatoInvalidoException si la línea está incompleta o es corrupta
     */
    public static Vehiculo deserialize(String linea) throws DatoInvalidoException {
        String[] campos = linea.split(SEPARADOR, -1);
        if (campos.length < CAMPOS_COMUNES) {
            throw new DatoInvalidoException("Registro CSV incompleto: " + campos.length + " campos.");
        }

        TipoVehiculo tipo = Validador.parsearEnum(campos[0], TipoVehiculo.class);
        String identificador = decodificar(campos[1]);
        String marca = decodificar(campos[2]);
        String modelo = decodificar(campos[3]);
        Integer anio = Validador.parsearEntero(campos[4], "anio");
        String color = decodificar(campos[5]);
        EstadoVehiculo estado = Validador.parsearEnum(campos[6], EstadoVehiculo.class);
        Double kilometraje = Validador.parsearDouble(campos[7], "kilometraje");
        Double precio = Validador.parsearDouble(campos[8], "precio");

        Vehiculo vehiculo = switch (tipo) {
            case AUTOMOVIL -> {
                verificarCantidad(campos, 11, "automóvil");
                yield new Automovil(identificador, marca, modelo, anio, color, estado, kilometraje,
                        precio, Validador.parsearEnum(campos[9], TipoCarroceria.class),
                        Validador.parsearEntero(campos[10], "puertas"));
            }
            case CAMIONETA -> {
                verificarCantidad(campos, 12, "camioneta");
                yield new Camioneta(identificador, marca, modelo, anio, color, estado, kilometraje,
                        precio, Validador.parsearEnum(campos[9], TipoCabina.class),
                        Validador.parsearEnum(campos[10], TipoTraccion.class),
                        Validador.parsearDouble(campos[11], "capacidadCarga"));
            }
            case MOTOCICLETA -> {
                verificarCantidad(campos, 11, "motocicleta");
                yield new Motocicleta(identificador, marca, modelo, anio, color, estado, kilometraje,
                        precio, Validador.parsearEnum(campos[9], EstiloMotocicleta.class),
                        Validador.parsearEntero(campos[10], "cilindrada"));
            }
        };

        vehiculo.validar();
        return vehiculo;
    }

    /**
     * Comprueba la cantidad exacta de campos según el tipo de vehículo.
     *
     * @param campos     arreglo de campos leídos
     * @param esperados  cantidad esperada
     * @param nombreTipo nombre del tipo para el mensaje
     * @throws DatoInvalidoException si la cantidad no coincide
     */
    private static void verificarCantidad(String[] campos, int esperados, String nombreTipo)
            throws DatoInvalidoException {
        if (campos.length != esperados) {
            throw new DatoInvalidoException("Registro de " + nombreTipo + " con "
                    + campos.length + " campos; se esperaban " + esperados + ".");
        }
    }

    /**
     * Codifica un texto a Base64 URL.
     *
     * @param texto texto original
     * @return texto codificado
     */
    private static String codificar(String texto) {
        return Base64.getUrlEncoder().encodeToString(texto.getBytes(StandardCharsets.UTF_8));
    }

    /**
     * Decodifica un campo Base64 URL.
     *
     * @param campo campo codificado
     * @return texto original
     * @throws DatoInvalidoException si el campo no es Base64 válido
     */
    private static String decodificar(String campo) throws DatoInvalidoException {
        try {
            return new String(Base64.getUrlDecoder().decode(campo), StandardCharsets.UTF_8);
        } catch (IllegalArgumentException causa) {
            throw new DatoInvalidoException("Campo de texto corrupto en el archivo CSV.", causa);
        }
    }
}

/**
 * Implementación genérica en memoria del contrato {@link IRepositorio}.
 *
 * <p><b>Decisión de diseño:</b> se usa un {@link LinkedHashMap} para indexar
 * por identificador (búsquedas O(1)) y conservar el orden de inserción en el
 * listado. La extracción del id se inyecta con una función
 * {@link Function}, de modo que esta clase sea reutilizable para cualquier entidad.</p>
 *
 * @param <T>  tipo de entidad gestionada
 * @param <ID> tipo del identificador
 */
public class RepositorioGenerico<T, ID> implements IRepositorio<T, ID> {

    /** Mapa indexado por identificador. */
    protected final Map<ID, T> entidades = new LinkedHashMap<>();

    /** Función que extrae el identificador de una entidad. */
    protected final Function<T, ID> extractorId;

    /**
     * Crea el repositorio indicando cómo obtener el identificador.
     *
     * @param extractorId función que extrae el id de la entidad
     */
    public RepositorioGenerico(Function<T, ID> extractorId) {
        this.extractorId = extractorId;
    }

    /**
     * {@inheritDoc}
     */
    @Override
    public void agregar(T entidad) throws VehiculoDuplicadoException {
        ID id = extractorId.apply(entidad);
        if (entidades.containsKey(id)) {
            throw new VehiculoDuplicadoException(String.valueOf(id));
        }
        entidades.put(id, entidad);
    }

    /**
     * {@inheritDoc}
     */
    @Override
    public T buscarPorId(ID id) throws VehiculoNoEncontradoException {
        T entidad = entidades.get(id);
        if (entidad == null) {
            throw new VehiculoNoEncontradoException(String.valueOf(id));
        }
        return entidad;
    }

    /**
     * {@inheritDoc}
     */
    @Override
    public List<T> listarTodos() {
        return new ArrayList<>(entidades.values());
    }

    /**
     * {@inheritDoc}
     */
    @Override
    public void actualizar(T entidad) throws VehiculoNoEncontradoException {
        ID id = extractorId.apply(entidad);
        if (!entidades.containsKey(id)) {
            throw new VehiculoNoEncontradoException(String.valueOf(id));
        }
        entidades.put(id, entidad);
    }

    /**
     * {@inheritDoc}
     */
    @Override
    public void eliminar(ID id) throws VehiculoNoEncontradoException {
        if (entidades.remove(id) == null) {
            throw new VehiculoNoEncontradoException(String.valueOf(id));
        }
    }

    /**
     * Reemplaza todo el contenido del repositorio, usado al cargar desde disco.
     *
     * @param nuevasEntidades colección de entidades a cargar
     */
    protected void reemplazarTodo(Collection<T> nuevasEntidades) {
        entidades.clear();
        for (T entidad : nuevasEntidades) {
            entidades.put(extractorId.apply(entidad), entidad);
        }
    }
}

        try (BufferedReader lector = Files.newBufferedReader(archivo, StandardCharsets.UTF_8)) {

            while ((linea = lector.readLine()) != null) {
                numeroLinea++;
                if (linea.isBlank()) {

/**

 *
 * <p>El servicio y la consola comparten estas reglas para que un mismo dato
 * se valide siempre de idéntica manera. Las conversiones usan los

public final class Validador {

    /** Constante para evitar instanciación de la clase utilitaria. */

    }

    /**
     * Valida que un texto no sea nulo ni esté vacío.
     *
     * @param texto       texto a validar
     * @param nombreCampo nombre del campo para el mensaje de error
     * @return texto recortado (sin espacios laterales)
     * @throws DatoInvalidoException si el texto es nulo o vacío
     */
    public static String validarTextoNoVacio(String texto, String nombreCampo)
            throws DatoInvalidoException {
        if (texto == null || texto.isBlank()) {
            throw new DatoInvalidoException("El campo '" + nombreCampo + "' no puede estar vacío.");
        }
        return texto.trim();
    }

    /**
     * Convierte un texto a {@link Integer} usando {@link Integer#valueOf(String)}.
     *
     * @param texto       texto a convertir
     * @param nombreCampo nombre del campo para el mensaje de error
     * @return valor entero parseado
     * @throws DatoInvalidoException si el texto no es un número entero válido
     */
    public static Integer parsearEntero(String texto, String nombreCampo)
            throws DatoInvalidoException {
        try {
            return Integer.valueOf(texto.trim());
        } catch (NumberFormatException causa) {
            throw new DatoInvalidoException(
                    "El campo '" + nombreCampo + "' debe ser un número entero.", causa);
        }
    }

    /**
     * Convierte un texto a {@link Double}, aceptando coma o punto decimal.
     *
     * @param texto       texto a convertir
     * @param nombreCampo nombre del campo para el mensaje de error
     * @return valor decimal parseado
     * @throws DatoInvalidoException si el texto no es un número válido
     */
    public static Double parsearDouble(String texto, String nombreCampo)
            throws DatoInvalidoException {
        try {
            return Double.valueOf(texto.trim().replace(',', '.'));
        } catch (NumberFormatException causa) {
            throw new DatoInvalidoException(
                    "El campo '" + nombreCampo + "' debe ser un número decimal.", causa);
        }
    }

    /**

    public static Integer validarAnio(Integer anio) throws DatoInvalidoException {
        if (anio == null || anio < Vehiculo.ANIO_MINIMO || anio > Vehiculo.ANIO_MAXIMO) {

    public static Double validarKilometraje(Double kilometraje) throws DatoInvalidoException {

    public static <E extends Enum<E>> E parsearEnum(String texto, Class<E> tipoEnum)

                    "Valor '" + texto + "' inválido para " + tipoEnum.getSimpleName() + ".", causa);
        }
    }

    /**
     * Formatea una lista de vehículos como una tabla alineada.
     *
     * @param vehiculos vehículos a mostrar
     * @return tabla lista para imprimir
     */
    public static String tabla(List<Vehiculo> vehiculos) {
        if (vehiculos.isEmpty()) {
            return "  No hay vehículos que coincidan con el criterio.";
        }
        StringBuilder salida = new StringBuilder();
        salida.append(String.format(Locale.US,"  %-16s %-14s %-14s %-6s %-13s %-12s %12s%n",
                "ID", "Marca", "Modelo", "Año", "Estado", "Tipo", "Precio"));
        salida.append("  ").append("-".repeat(94)).append(System.lineSeparator());
        for (Vehiculo vehiculo : vehiculos) {
            salida.append(String.format(Locale.US,"  %-16s %-14s %-14s %-6d %-13s %-12s %12.2f%n",
                    cortar(vehiculo.getIdentificador(), 16),
                    cortar(vehiculo.getMarca(), 14),
                    cortar(vehiculo.getModelo(), 14),
                    vehiculo.getAnio(),
                    vehiculo.getEstado().getEtiqueta(),
                    vehiculo.obtenerTipo().getEtiqueta(),
                    vehiculo.getPrecio()));
        }
        salida.append("  Total de registros: ").append(vehiculos.size());
        return salida.toString();
    }

     * @param vehiculo vehículo a mostrar

     * @return resumen formateado

                + "  TOTAL A COBRAR: $" + String.format(Locale.US,"%.2f", simulacion.getTotal());
    }

    /**
     * Recorta un texto para que no rompa el ancho de la columna.
     *
     * @param texto texto original
     * @param ancho ancho máximo
     * @return texto recortado
     */
    private static String cortar(String texto, int ancho) {
        return texto.length() <= ancho ? texto : texto.substring(0, ancho - 1) + "…";
    }

    /**
     * Escapa una cadena para usarla dentro de un literal JSON.
     *
     * @param texto cadena original
     * @return cadena escapada, sin las comillas envolventes
     */
    public static String escape(String texto) {
        StringBuilder salida = new StringBuilder();
        for (int i = 0; i < texto.length(); i++) {
            char caracter = texto.charAt(i);
            switch (caracter) {
                case '"' -> salida.append("\\\"");
                case '\\' -> salida.append("\\\\");
                case '\b' -> salida.append("\\b");
                case '\f' -> salida.append("\\f");
                case '\n' -> salida.append("\\n");
                case '\r' -> salida.append("\\r");
                case '\t' -> salida.append("\\t");
                default -> {
                    if (caracter < 0x20) {
                        salida.append(String.format("\\u%04x", (int) caracter));
                    } else {
                        salida.append(caracter);
                    }
                }
            }
        }
        return salida.toString();
    }

    /**
     * Atiende una petición HTTP y envía la respuesta.
     *
     * @param cliente socket del cliente
     */
    private void atender(Socket cliente) {
        try (cliente;
             BufferedReader lector = new BufferedReader(
                     new InputStreamReader(cliente.getInputStream(), StandardCharsets.UTF_8));
             OutputStream salida = cliente.getOutputStream()) {

            String solicitud = lector.readLine();
            if (solicitud == null || solicitud.isBlank()) {
                return;
            }
            String[] partes = solicitud.split(" ");
            if (partes.length < 2) {
                return;
            }
            String metodo = partes[0].toUpperCase();
            String objetivo = partes[1];

            Map<String, String> cabeceras = new HashMap<>();
            String cabecera;
            while ((cabecera = lector.readLine()) != null && !cabecera.isEmpty()) {
                int separador = cabecera.indexOf(':');
                if (separador > 0) {
                    cabeceras.put(cabecera.substring(0, separador).trim().toLowerCase(),
                            cabecera.substring(separador + 1).trim());
                }
            }

            Integer longitud = 0;
            try {
                longitud = Integer.parseInt(cabeceras.getOrDefault("content-length", "0"));
            } catch (NumberFormatException ignorado) {
                longitud = 0;
            }
            char[] cuerpoCaracteres = new char[longitud];
            Integer leidos = 0;
            while (leidos < longitud) {
                int cantidad = lector.read(cuerpoCaracteres, leidos, longitud - leidos);
                if (cantidad < 0) {
                    break;
                }
                leidos += cantidad;
            }
            String cuerpo = new String(cuerpoCaracteres, 0, leidos);

            Respuesta respuesta = enrutar(metodo, objetivo, cuerpo);
            escribir(salida, respuesta);
        } catch (IOException error) {
            System.out.println("Error en petición: " + error.getMessage());
        } catch (RuntimeException error) {
            System.out.println("Error inesperado en petición: " + error.getMessage());
        }
    }

    /**
     * Dirige la petición al controlador API o al archivo estático correspondiente.
     *
     * @param metodo    método HTTP
     * @param objetivo  ruta con query string
     * @param cuerpo    cuerpo de la petición
     * @return respuesta HTTP
     */
    private Respuesta enrutar(String metodo, String objetivo, String cuerpo) {
        try {
            int indiceConsulta = objetivo.indexOf('?');
            String ruta = indiceConsulta >= 0 ? objetivo.substring(0, indiceConsulta) : objetivo;
            String consulta = indiceConsulta >= 0 ? objetivo.substring(indiceConsulta + 1) : "";

            Map<String, String> parametros = new LinkedHashMap<>();
            parametros.putAll(parsearFormulario(consulta));
            parametros.putAll(parsearFormulario(cuerpo));

            if (ruta.startsWith("/api/")) {
                return api(metodo, ruta, parametros);
            }
            if (metodo.equals("GET") || metodo.equals("HEAD")) {
                return estatico(ruta);
            }
            return new Respuesta(405, Json.error("Método no permitido."), TIPO_JSON);
        } catch (DatoInvalidoException | VehiculoDuplicadoException
                 | VehiculoNoEncontradoException error) {
            return new Respuesta(400, Json.error(error.getMessage()), TIPO_JSON);
        } catch (NumberFormatException error) {
            return new Respuesta(400, Json.error("Se esperaba un valor numérico."), TIPO_JSON);
        } catch (IOException error) {
            return new Respuesta(500, Json.error("Error de persistencia: " + error.getMessage()),
                    TIPO_JSON);
        }
    }

    /**
     * Controlador de la API REST.
     *
     * @param metodo     método HTTP
     * @param ruta       ruta de la API
     * @param parametros parámetros de consulta y cuerpo
     * @return respuesta HTTP
     * @throws DatoInvalidoException         si un parámetro no es válido
     * @throws VehiculoDuplicadoException    si se repite un identificador
     * @throws VehiculoNoEncontradoException si no existe el vehículo
     * @throws IOException                   si falla la persistencia
     */
    private Respuesta api(String metodo, String ruta, Map<String, String> parametros)
            throws DatoInvalidoException, VehiculoDuplicadoException,
            VehiculoNoEncontradoException, IOException {

        if (ruta.equals("/api/opciones") && metodo.equals("GET")) {
            return new Respuesta(200, Json.opciones(), TIPO_JSON);
        }

        if (ruta.equals("/api/vehiculos")) {
            if (metodo.equals("GET")) {
                TipoVehiculo tipo = enumOpcional(parametros.get("tipo"), TipoVehiculo.class);
                EstadoVehiculo estado = enumOpcional(parametros.get("estado"), EstadoVehiculo.class);
                List<Vehiculo> resultados = servicio.listarConFiltros(tipo, estado,
                        parametros.get("marca"));
                return new Respuesta(200, Json.lista(resultados), TIPO_JSON);
            }
            if (metodo.equals("POST")) {
                Vehiculo vehiculo = construirVehiculo(parametros);
                servicio.darDeAlta(vehiculo);
                return new Respuesta(201, Json.mensaje("Vehículo dado de alta correctamente."),
                        TIPO_JSON);
            }
            return new Respuesta(405, Json.error("Método no permitido."), TIPO_JSON);
        }

        if (ruta.equals("/api/simulacion") && metodo.equals("POST")) {
            SimulacionVenta simulacion = servicio.simularVenta(
                    parametros.get("id"),
                    Validador.parsearDouble(parametros.get("descuento"), "descuento"));
            return new Respuesta(200, Json.simulacion(simulacion), TIPO_JSON);
        }

        if (ruta.startsWith("/api/vehiculos/")) {
            String resto = ruta.substring("/api/vehiculos/".length());
            String id = URLDecoder.decode(resto.split("/")[0], StandardCharsets.UTF_8);
            if (metodo.equals("PUT") && resto.endsWith("/precio")) {
                servicio.cambiarPrecio(id, Validador.parsearDouble(parametros.get("precio"), "precio"));
                return new Respuesta(200, Json.mensaje("Precio actualizado."), TIPO_JSON);
            }
            if (metodo.equals("PUT") && resto.endsWith("/color")) {
                servicio.cambiarColor(id, parametros.get("color"));
                return new Respuesta(200, Json.mensaje("Color actualizado."), TIPO_JSON);
            }
            if (metodo.equals("DELETE") && !resto.contains("/")) {
                servicio.darDeBaja(id);
                return new Respuesta(200, Json.mensaje("Vehículo eliminado."), TIPO_JSON);
            }
        }

        return new Respuesta(404, Json.error("Recurso no encontrado."), TIPO_JSON);
    }

    /**
     * Convierte un parámetro opcional a enum.
     *
     * @param valor     texto del parámetro, puede ser {@code null} o vacío
     * @param tipoEnum  clase del enum
     * @param <E>       tipo del enum

    /**
     * Sirve un archivo estático protegiendo contra path traversal.
     *
     * @param ruta ruta solicitada
     * @return respuesta HTTP con el archivo o 404
     * @throws IOException si falla la lectura
     */
    private Respuesta estatico(String ruta) throws IOException {
        String nombre = ruta.equals("/") ? "index.html" : ruta.substring(1);
        Path archivo = raizEstatica.resolve(nombre).normalize();
        if (!archivo.startsWith(raizEstatica) || !Files.isRegularFile(archivo)) {
            return new Respuesta(404, Json.error("Recurso no encontrado."), TIPO_JSON);
        }
        String contenido = new String(Files.readAllBytes(archivo), StandardCharsets.UTF_8);
        return new Respuesta(200, contenido, tipoContenido(archivo));
    }

    /**
     * Devuelve el tipo MIME según la extensión del archivo.
     *
     * @param archivo archivo solicitado
     * @return tipo de contenido HTTP
     */
    private String tipoContenido(Path archivo) {
        String nombre = archivo.getFileName().toString().toLowerCase();
        if (nombre.endsWith(".html")) {
            return "text/html; charset=utf-8";
        }
        if (nombre.endsWith(".css")) {
            return "text/css; charset=utf-8";
        }
        if (nombre.endsWith(".js")) {
            return "application/javascript; charset=utf-8";
        }
        if (nombre.endsWith(".svg")) {
            return "image/svg+xml";
        }
        return "application/octet-stream";
    }

    /**
     * Analiza un cuerpo con formato application/x-www-form-urlencoded.
     *
     * @param texto texto a analizar
     * @return mapa de parámetros
     */
    private Map<String, String> parsearFormulario(String texto) {
        Map<String, String> parametros = new LinkedHashMap<>();
        if (texto == null || texto.isEmpty()) {
            return parametros;
        }
        for (String pareja : texto.split("&")) {
            if (pareja.isEmpty()) {
                continue;
            }
            String[] partes = pareja.split("=", 2);
            try {
                String clave = URLDecoder.decode(partes[0], StandardCharsets.UTF_8);
                String valor = partes.length > 1
                        ? URLDecoder.decode(partes[1], StandardCharsets.UTF_8)
                        : "";
                parametros.put(clave, valor);
            } catch (IllegalArgumentException ignorado) {
                // Se ignora un parámetro mal codificado; el resto sigue siendo válido.
            }
        }
        return parametros;
    }

    /**
     * Escribe la respuesta HTTP con cabeceras y cuerpo.
     *
     * @param salida   flujo de salida del socket
     * @param respuesta respuesta a enviar
     * @throws IOException si falla la escritura
     */
    private void escribir(OutputStream salida, Respuesta respuesta) throws IOException {
        byte[] cuerpo = respuesta.contenido.getBytes(StandardCharsets.UTF_8);
        String cabeceras = "HTTP/1.1 " + respuesta.codigo + " "
                + razon(respuesta.codigo) + "\r\n"
                + "Content-Type: " + respuesta.tipo + "\r\n"
                + "Content-Length: " + cuerpo.length + "\r\n"
                + "Cache-Control: no-store\r\n"
                + "Connection: close\r\n\r\n";
        salida.write(cabeceras.getBytes(StandardCharsets.US_ASCII));
        salida.write(cuerpo);
        salida.flush();
    }

    /**
     * Devuelve el texto descriptivo de un código HTTP.
     *
     * @param codigo código HTTP
     * @return razón del código
     */
    private String razon(Integer codigo) {
        return switch (codigo) {
            case 200 -> "OK";
            case 201 -> "Created";
            case 400 -> "Bad Request";
            case 404 -> "Not Found";
            case 405 -> "Method Not Allowed";
            default -> "Internal Server Error";
        };
    }

     * @param args argumentos opcionales: puerto y directorio de archivos estáticos

    public static void main(String[] args) {

        Path estatico = ESTATICO_DEFECTO;
        try {
            if (args.length > 0) {
                puerto = Integer.valueOf(args[0]);
            }
            if (args.length > 1) {
                estatico = Paths.get(args[1]);
            }
            new ServidorWeb(puerto, estatico).iniciar();
        } catch (NumberFormatException error) {
            System.out.println("El puerto debe ser un número entero.");
        } catch (DatoInvalidoException | IOException error) {
            System.out.println("No se pudo iniciar el servidor: " + error.getMessage());
        }
    }

    /**
     * Respuesta HTTP simple.

    private static final class Respuesta {

        private final String tipo;

        private Respuesta(Integer codigo, String contenido, String tipo) {
            this.codigo = codigo;
            this.contenido = contenido;
            this.tipo = tipo;
        }

function llenarSelect(select, opciones, conservarPrimera = false) {
  const primera = conservarPrimera ? select.options[0].outerHTML : '';
  select.innerHTML = primera + opciones.map((opcion) =>
    `<option value="${esc(opcion.valor)}">${esc(opcion.etiqueta)}</option>`).join('');
}

/** Carga los enums expuestos por el servidor. */
async function cargarOpciones() {
  const opciones = await api('/api/opciones');
  llenarSelect($('filtroTipo'), opciones.tipos, true);
  llenarSelect($('filtroEstado'), opciones.estados, true);
  llenarSelect($('tipo'), opciones.tipos);
  llenarSelect($('estado'), opciones.estados);
  llenarSelect($('carroceria'), opciones.carrocerias);
  llenarSelect($('cabina'), opciones.cabinas);
  llenarSelect($('traccion'), opciones.tracciones);
  llenarSelect($('estilo'), opciones.estilos);
  actualizarCamposEspecificos();
}

/** Muestra solo los campos correspondientes al tipo y estado elegidos. */
function actualizarCamposEspecificos() {
  const tipo = $('tipo').value;
  $('campoCarroceria').hidden = tipo !== 'AUTOMOVIL';
  $('campoPuertas').hidden = tipo !== 'AUTOMOVIL';
  $('campoCabina').hidden = tipo !== 'CAMIONETA';
  $('campoTraccion').hidden = tipo !== 'CAMIONETA';
  $('campoCarga').hidden = tipo !== 'CAMIONETA';
  $('campoEstilo').hidden = tipo !== 'MOTOCICLETA';
  $('campoCilindrada').hidden = tipo !== 'MOTOCICLETA';
  $('campoKm').hidden = $('estado').value !== 'USADO';
}

/** Consulta el stock aplicando los filtros de la pantalla. */
async function listar() {
  const parametros = new URLSearchParams();
  if ($('filtroTipo').value) parametros.set('tipo', $('filtroTipo').value);
  if ($('filtroEstado').value) parametros.set('estado', $('filtroEstado').value);
  if ($('filtroMarca').value.trim()) parametros.set('marca', $('filtroMarca').value.trim());
  estado.vehiculos = await api('/api/vehiculos?' + parametros.toString());
  pintar();
}

/** Actualiza tarjetas, tabla y selector de simulación. */
function pintar() {
  const lista = estado.vehiculos;
  const nuevos = lista.filter((v) => v.estado === 'NUEVO').length;
  const valor = lista.reduce((suma, v) => suma + Number(v.precio), 0);
  $('statTotal').textContent = lista.length;
  $('statNuevos').textContent = nuevos;
  $('statUsados').textContent = lista.length - nuevos;
  $('statValor').textContent = dinero(valor);
  $('contador').textContent = lista.length + ' resultado(s)';

  if (lista.length === 0) {
    $('cuerpoTabla').innerHTML =
      '<tr><td colspan="9" class="vacio">No hay vehículos que coincidan con los filtros.</td></tr>';
  } else {
    $('cuerpoTabla').innerHTML = lista.map((v) => `
      <tr>
        <td><strong>${esc(v.id)}</strong></td>
        <td>${esc(v.marca)} ${esc(v.modelo)}</td>
        <td>${esc(v.anio)}</td>
        <td>${esc(v.color)}</td>
        <td><span class="pill ${v.estado === 'NUEVO' ? 'nuevo' : 'usado'}">${esc(v.estadoEtiqueta)}</span></td>
        <td>${esc(v.tipoEtiqueta)}</td>
        <td class="precio">${dinero(v.precio)}</td>
        <td>${dinero(v.impuesto)}</td>
        <td>
          <div class="acciones-tabla">
            <button class="btn pequeño" data-accion="ficha" data-id="${esc(v.id)}">Ficha</button>
            <button class="btn pequeño" data-accion="editar" data-id="${esc(v.id)}">Editar</button>
            <button class="btn pequeño" data-accion="simular" data-id="${esc(v.id)}">Simular</button>
            <button class="btn pequeño peligro" data-accion="eliminar" data-id="${esc(v.id)}">Baja</button>
          </div>
        </td>
      </tr>`).join('');
  }

  $('simularId').innerHTML = lista.map((v) =>
    `<option value="${esc(v.id)}">${esc(v.id)} — ${esc(v.marca)} ${esc(v.modelo)}</option>`).join('');
}

/** Abre un modal por id. */
function abrir(id) { $(id).hidden = false; }

/** Cierra un modal por id. */
function cerrar(id) { $(id).hidden = true; }

/** Configura los eventos de la interfaz. */
function cablear() {
  $('btnNuevo').addEventListener('click', () => abrir('modalAlta'));
  $('btnSimular').addEventListener('click', () => {
    $('resultadoSimulacion').hidden = true;
    abrir('modalSimular');
  });
  $('btnRecargar').addEventListener('click', async () => {
    try { await listar(); aviso('Stock actualizado.'); } catch (error) { aviso(error.message, true); }
  });
  $('btnFiltrar').addEventListener('click', async () => {
    try { await listar(); } catch (error) { aviso(error.message, true); }
  });
  $('btnLimpiar').addEventListener('click', async () => {
    $('filtroTipo').value = '';
    $('filtroEstado').value = '';
    $('filtroMarca').value = '';
    try { await listar(); } catch (error) { aviso(error.message, true); }
  });

  $('tipo').addEventListener('change', actualizarCamposEspecificos);
  $('estado').addEventListener('change', actualizarCamposEspecificos);

  document.querySelectorAll('[data-cerrar]').forEach((boton) => {
    boton.addEventListener('click', () => cerrar(boton.dataset.cerrar));
  });
  document.querySelectorAll('.modal').forEach((modal) => {
    modal.addEventListener('click', (evento) => {
      if (evento.target === modal) modal.hidden = true;
    });
  });

  $('cuerpoTabla').addEventListener('click', (evento) => {
    const boton = evento.target.closest('button[data-accion]');
    if (!boton) return;
    const vehiculo = estado.vehiculos.find((v) => v.id === boton.dataset.id);
    switch (boton.dataset.accion) {
      case 'ficha':
        $('fichaTexto').textContent = vehiculo.ficha;
        abrir('modalFicha');
        break;
      case 'editar':
        $('editarId').value = vehiculo.id;
        $('editarFicha').textContent = vehiculo.ficha;
        $('editarPrecio').value = vehiculo.precio;
        $('editarColor').value = vehiculo.color;
        abrir('modalEditar');
        break;
      case 'simular':
        $('simularId').value = vehiculo.id;
        $('resultadoSimulacion').hidden = true;
        abrir('modalSimular');
        break;
      case 'eliminar':
        eliminar(vehiculo);
        break;
    }
  });

  $('formAlta').addEventListener('submit', async (evento) => {
    evento.preventDefault();
    const datos = Object.fromEntries(new FormData(evento.target).entries());
    try {
      await api('/api/vehiculos', { method: 'POST', headers: CABECERA, body: cuerpo(datos) });
      cerrar('modalAlta');
      evento.target.reset();
      actualizarCamposEspecificos();
      await listar();
      aviso('Vehículo dado de alta correctamente.');
    } catch (error) {
      aviso(error.message, true);
    }
  });

  $('formEditar').addEventListener('submit', async (evento) => {
    evento.preventDefault();
    const id = encodeURIComponent($('editarId').value);
    try {
      const precio = $('editarPrecio').value;
      const color = $('editarColor').value.trim();
      if (precio) {
        await api('/api/vehiculos/' + id + '/precio',
          { method: 'PUT', headers: CABECERA, body: cuerpo({ precio }) });
      }
      if (color) {
        await api('/api/vehiculos/' + id + '/color',
          { method: 'PUT', headers: CABECERA, body: cuerpo({ color }) });
      }
      cerrar('modalEditar');
      await listar();
      aviso('Modificación guardada.');
    } catch (error) {
      aviso(error.message, true);
    }
  });

  $('formSimular').addEventListener('submit', async (evento) => {
    evento.preventDefault();
    try {
      const resultado = await api('/api/simulacion', {
        method: 'POST', headers: CABECERA,
        body: cuerpo({ id: $('simularId').value, descuento: $('simularDescuento').value })
      });
      $('resultadoSimulacion').innerHTML = `
        <div><span>Precio base</span><strong>${dinero(resultado.precioBase)}</strong></div>
        <div><span>Descuento (${resultado.descuento}%)</span><strong>- ${dinero(resultado.precioBase - resultado.precioFinal)}</strong></div>
        <div><span>Precio final</span><strong>${dinero(resultado.precioFinal)}</strong></div>
        <div><span>Impuesto de patente</span><strong>${dinero(resultado.impuesto)}</strong></div>
        <div><span>Valor tasado</span><strong>${dinero(resultado.tasado)}</strong></div>
        <div class="total"><span>Total a cobrar</span><strong>${dinero(resultado.total)}</strong></div>`;
      $('resultadoSimulacion').hidden = false;
    } catch (error) {
      aviso(error.message, true);
    }
  });
}

/** Da de baja un vehículo previa confirmación. */
async function eliminar(vehiculo) {
  if (!confirm('¿Confirmar la baja de ' + vehiculo.id + ' — ' + vehiculo.marca + ' ' + vehiculo.modelo + '?')) return;
  try {
    await api('/api/vehiculos/' + encodeURIComponent(vehiculo.id), { method: 'DELETE' });
    await listar();
    aviso('Vehículo eliminado.');
  } catch (error) {
    aviso(error.message, true);
  }
}

/** Arranque de la aplicación. */
async function iniciar() {
  cablear();
  try {
    await cargarOpciones();
    await listar();
  } catch (error) {
    aviso('No se pudo conectar con el servidor: ' + error.message, true);
  }
}
